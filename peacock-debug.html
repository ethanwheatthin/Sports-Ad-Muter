🦚 Peacock Video Debug Tool

Instructions

1. Open Peacock in another tab

2. Start playing a video

3. Open browser DevTools (F12)

4. Paste the script below into the Console

5. Copy the output and analyze

Debug Script

// Peacock Video Debug Script
(function() {
    console.log('=== PEACOCK VIDEO DEBUG ===');
    console.log('Timestamp:', new Date().toISOString());
    console.log('');
    
    // Find all video elements
    const videos = document.querySelectorAll('video');
    console.log('Total video elements found:', videos.length);
    console.log('');
    
    // Analyze each video
    videos.forEach((video, index) => {
        console.log(`--- VIDEO ${index + 1} ---`);
        console.log('Properties:');
        console.log('  videoWidth:', video.videoWidth);
        console.log('  videoHeight:', video.videoHeight);
        console.log('  area:', video.videoWidth * video.videoHeight);
        console.log('  currentTime:', video.currentTime);
        console.log('  duration:', video.duration);
        console.log('  paused:', video.paused);
        console.log('  muted:', video.muted);
        console.log('  readyState:', video.readyState, 
            ['HAVE_NOTHING', 'HAVE_METADATA', 'HAVE_CURRENT_DATA', 'HAVE_FUTURE_DATA', 'HAVE_ENOUGH_DATA'][video.readyState]);
        console.log('  networkState:', video.networkState,
            ['NETWORK_EMPTY', 'NETWORK_IDLE', 'NETWORK_LOADING', 'NETWORK_NO_SOURCE'][video.networkState]);
        console.log('  src:', video.src || video.currentSrc || 'no src');
        console.log('  crossOrigin:', video.crossOrigin);
        
        // Check parent elements
        console.log('Parent chain:');
        let parent = video.parentElement;
        let depth = 0;
        while (parent && depth < 5) {
            const classes = parent.className || 'no classes';
            const id = parent.id || 'no id';
            const tag = parent.tagName.toLowerCase();
            console.log(`  ${depth + 1}. <${tag}> id="${id}" class="${classes}"`);
            parent = parent.parentElement;
            depth++;
        }
        
        // Check visibility
        const rect = video.getBoundingClientRect();
        const style = window.getComputedStyle(video);
        console.log('Visibility:');
        console.log('  display:', style.display);
        console.log('  visibility:', style.visibility);
        console.log('  opacity:', style.opacity);
        console.log('  position:', style.position);
        console.log('  zIndex:', style.zIndex);
        console.log('  top:', rect.top, 'left:', rect.left);
        console.log('  width:', rect.width, 'height:', rect.height);
        console.log('  inViewport:', rect.top >= 0 && rect.left >= 0 && 
            rect.bottom <= window.innerHeight && rect.right <= window.innerWidth);
        
        // Check if video is actually playing
        console.log('Playback state:');
        console.log('  playing:', !video.paused && !video.ended && video.readyState > 2);
        console.log('  ended:', video.ended);
        console.log('  seeking:', video.seeking);
        console.log('  buffered:', video.buffered.length > 0 ? 
            `${video.buffered.start(0)} - ${video.buffered.end(video.buffered.length - 1)}` : 'none');
        
        // Try to capture frame
        console.log('Capture test:');
        try {
            const canvas = document.createElement('canvas');
            canvas.width = Math.min(video.videoWidth, 100);
            canvas.height = Math.min(video.videoHeight, 100);
            const ctx = canvas.getContext('2d');
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
            
            // Check pixel variance
            let variance = 0;
            for (let i = 0; i < imageData.data.length; i += 16) {
                const r = imageData.data[i];
                const g = imageData.data[i + 1];
                const b = imageData.data[i + 2];
                if ((r > 5 || g > 5 || b > 5) && (r < 250 || g < 250 || b < 250)) {
                    variance++;
                }
            }
            
            console.log('  ✅ Capture successful');
            console.log('  Pixel variance:', variance);
            console.log('  Sample size:', Math.floor(imageData.data.length / 16));
            console.log('  Has content:', variance > 5);
        } catch (e) {
            console.log('  ❌ Capture failed:', e.message);
        }
        
        console.log('');
    });
    
    // Check for Peacock-specific selectors
    console.log('--- PEACOCK SPECIFICS ---');
    const peacockSelectors = [
        '.video-player',
        '.player-container',
        '[class*="video"]',
        '[class*="player"]',
        '[data-testid*="video"]',
        '[data-testid*="player"]',
        'video[class]',
        'video[id]'
    ];
    
    peacockSelectors.forEach(selector => {
        try {
            const elements = document.querySelectorAll(selector);
            if (elements.length > 0) {
                console.log(`Found ${elements.length} element(s) matching: ${selector}`);
                elements.forEach((el, i) => {
                    if (el.tagName === 'VIDEO') {
                        console.log(`  ${i + 1}. VIDEO element`);
                    } else {
                        const videoInside = el.querySelector('video');
                        console.log(`  ${i + 1}. ${el.tagName} ${videoInside ? 'WITH video inside' : 'NO video'}`);
                    }
                });
            }
        } catch (e) {
            // Invalid selector
        }
    });
    
    console.log('');
    console.log('=== END DEBUG ===');
})();

Common Issues with Peacock

  • Multiple video elements: Peacock may use multiple video tags (one for ads, one for content)
  • Dynamic loading: Video elements may be created/destroyed dynamically
  • Hidden videos: Inactive videos may have 0x0 dimensions or be off-screen
  • CORS restrictions: Some streaming services block cross-origin canvas access
  • DRM/EME: Protected content may have additional restrictions

What to Look For in Output

  1. Number of videos: Are there multiple video elements?
  2. Dimensions: Which video has the largest dimensions (width × height)?
  3. Playing state: Which video is actually playing (not paused)?
  4. ReadyState: Should be 3 or 4 for active playback
  5. Capture test: Does the capture succeed or fail?
  6. Pixel variance: Should be > 5 for real content
  7. Parent classes: Look for identifying classes like "active", "playing", "main"
